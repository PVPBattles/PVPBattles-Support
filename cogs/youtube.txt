def sanitize_channel_name(name):
    name = str(name).lower().replace(" ", "-")
    name = "".join(
        c for c in name
        if c in "abcdefghijklmnopqrstuvwxyz0123456789-_"
    )
    return name[:90] or "ticket"


def replace_placeholders(text, member, category=None):
    replacements = {
        "{username}": member.name,
        "{displayname}": member.display_name,
        "{userid}": str(member.id),
        "{user}": member.mention,
        "{category}": category or "",
    }

    text = str(text)

    for key, value in replacements.items():
        text = text.replace(key, value)

    return text


def state(topic):
    creator = None
    replied = False

    if topic:
        for part in topic.split(";"):
            if part.startswith("ticket_creator:"):
                value = part.split(":", 1)[1]

                if value.isdigit():
                    creator = int(value)

            if part == "first_replied:1":
                replied = True

    return creator, replied


def topic(creator, replied=False):
    return (
        f"ticket_creator:{creator};"
        f"first_replied:{1 if replied else 0}"
    )


def staff(member, ids):
    valid_ids = {
        int(x)
        for x in ids
        if str(x).isdigit()
    }

    return any(role.id in valid_ids for role in member.roles)


def categories(config):
    value = config.get(
        "ticket_support_categories",
        []
    )

    if not isinstance(value, list):
        return []

    result = []

    for item in value:
        if isinstance(item, str):
            item = item.strip()

            if item:
                result.append(item[:100])

    return result[:25]


def auto_answers(config):
    value = config.get(
        "ticket_auto_answers",
        []
    )

    if not isinstance(value, list):
        return []

    result = []

    for item in value:
        if not isinstance(item, dict):
            continue

        keyword = str(
            item.get("keyword", "")
        ).strip()

        response = str(
            item.get("response", "")
        ).strip()

        if keyword and response:
            result.append({
                "keyword": keyword[:100],
                "response": response[:2000],
            })

    return result[:25]

def color(value):
    try:
        return discord.Color(
            int(
                str(value)
                .replace("#", ""),
                16
            )
        )
    except Exception:
        return discord.Color(0x58A6FF)

# Ticket creation
# =========================================================

async def create_ticket(
    interaction,
    support_category=None
):
    if (
        not interaction.guild
        or not isinstance(
            interaction.user,
            discord.Member
        )
    ):
        await interaction.response.send_message(
            "❌ サーバー内でのみ使用できます。",
            ephemeral=True
        )
        return

    config = cfg(interaction.guild.id)

    try:
        category = interaction.guild.get_channel(
            int(config.get("ticket_category_id"))
        )
    except (TypeError, ValueError):
        category = None

    if not isinstance(
        category,
        discord.CategoryChannel
    ):
        await interaction.response.send_message(
            "❌ チケットを作成するDiscordカテゴリが設定されていません。",
            ephemeral=True
        )
        return

    # 重複チケット防止
    for channel in category.text_channels:
        creator, _ = state(channel.topic)

        if creator == interaction.user.id:
            await interaction.response.send_message(
                "❌ すでにチケットがあります。\n"
                f"{channel.mention}",
                ephemeral=True
            )
            return

    overwrites = {
        interaction.guild.default_role:
            discord.PermissionOverwrite(
                view_channel=False
            ),

        interaction.user:
            discord.PermissionOverwrite(
                view_channel=True,
                send_messages=True,
                read_message_history=True,
                attach_files=True,
                embed_links=True
            ),
    }

    # Bot権限
    if interaction.guild.me:
        overwrites[interaction.guild.me] = (
            discord.PermissionOverwrite(
                view_channel=True,
                send_messages=True,
                read_message_history=True,
                manage_channels=True,
                manage_messages=True
            )
        )

    mentions = []

    role_ids = config.get(
        "ticket_staff_role_ids",
        []
    )

    if isinstance(role_ids, list):
        for role_id in role_ids:
            try:
                role = interaction.guild.get_role(
                    int(role_id)
                )
            except (TypeError, ValueError):
                role = None

            if role:
                overwrites[role] = (
                    discord.PermissionOverwrite(
                        view_channel=True,
                        send_messages=True,
                        read_message_history=True,
                        attach_files=True,
                        embed_links=True
                    )
                )

                mentions.append(role.mention)

    channel_name = sanitize_channel_name(
        replace_placeholders(
            config["ticket_name"],
            interaction.user,
            support_category
        )
    )

    await interaction.response.defer(
        ephemeral=True
    )

    try:
        channel = await interaction.guild.create_text_channel(
            name=channel_name,
            category=category,
            overwrites=overwrites,
            topic=topic(
                interaction.user.id
            ),
            reason=(
                f"Ticket created by "
                f"{interaction.user}"
            )
        )

        parts = []

        if mentions:
            parts.append(
                " ".join(mentions)
            )

        if support_category:
            parts.append(
                f"📂 **お問い合わせカテゴリ:** "
                f"{support_category}"
            )

        parts.append(
            replace_placeholders(
                config["ticket_greeting"],
                interaction.user,
                support_category
            )
        )

        await channel.send(
            "\n\n".join(parts),
            view=TicketCloseView(config)
        )

        await interaction.followup.send(
            "✅ チケットを作成しました！\n"
            f"{channel.mention}",
            ephemeral=True
        )

    except discord.HTTPException:
        await interaction.followup.send(
            "❌ チケットの作成に失敗しました。",
            ephemeral=True
        )

class SupportSelect(discord.ui.Select):

    def __init__(self, items):
        options = [
            discord.SelectOption(
                label=item[:100],
                value=str(index),
                emoji="📂"
            )
            for index, item in enumerate(items)
        ]

        super().__init__(
            placeholder="お問い合わせ内容を選択してください",
            options=options,
            min_values=1,
            max_values=1
        )

    async def callback(self, interaction):

        items = categories(
            get_guild_config(
                interaction.guild.id
            )
        )

        try:
            index = int(self.values[0])
            item = items[index]

        except (ValueError, IndexError):
            await interaction.response.send_message(
                "❌ カテゴリの取得に失敗しました。",
                ephemeral=True
            )
            return

        await create_ticket(
            interaction,
            item
        )


class SupportSelectView(discord.ui.View):

    def __init__(self, items):
        super().__init__(timeout=120)
        self.add_item(
            SupportSelect(items)
        )

# =========================================================

class TicketCloseView(discord.ui.View):

    def __init__(self, config=None):
        super().__init__(timeout=None)

        label = str(
            (config or {}).get(
                "ticket_close_label",
                "チケットを閉じる"
            )
        )[:80]

        if not label:
            label = "チケットを閉じる"

        button = discord.ui.Button(
            label=label,
            emoji="🔒",
            style=discord.ButtonStyle.danger,
            custom_id=TICKET_CLOSE_ID
        )

        button.callback = self.close

        self.add_item(button)

    async def close(self, interaction):

        if (
            not interaction.guild
            or not isinstance(
                interaction.user,
                discord.Member
            )
            or not isinstance(
                interaction.channel,
                discord.TextChannel
            )
        ):
            await interaction.response.send_message(
                "❌ このチャンネルでは使用できません。",
                ephemeral=True
            )
            return

        creator, _ = state(
            interaction.channel.topic
        )

        config = get_guild_config(
            interaction.guild.id
        )

        is_staff_member = staff(
            interaction.user,
            config.get(
                "ticket_staff_role_ids",
                []
            )
        )

        if not is_staff_member:

            if creator == interaction.user.id:
                await interaction.response.send_message(
                    "❌ チケットを作成した本人でも、"
                    "スタッフロールがない場合は閉じられません。",
                    ephemeral=True
                )
                return

            await interaction.response.send_message(
                "❌ このチケットを閉じられるのはスタッフのみです。",
                ephemeral=True
            )
            return

        await interaction.response.send_message(
            "🔒 チケットを閉じています……",
            ephemeral=True
        )

        try:
            await interaction.channel.delete(
                reason=f"Ticket closed by {interaction.user}"
            )
        except discord.HTTPException:
            pass

class TextModal(discord.ui.Modal):

    def __init__(
        self,
        title,
        field,
        label,
        value,
        max_length=4000
    ):
        super().__init__(
            title=title
        )

        self.field = field

        self.input = discord.ui.TextInput(
            label=label,
            default=str(value)[:max_length],
            max_length=max_length,
            required=True,
            style=discord.TextStyle.paragraph
        )

        self.add_item(
            self.input
        )

    async def on_submit(self, interaction):

        value = str(
            self.input
        ).strip()

        if not value:
            await interaction.response.send_message(
                "❌ 内容を入力してください。",
                ephemeral=True
            )
            return

        set_guild_value(
            interaction.guild.id,
            self.field,
            value
        )

        await interaction.response.send_message(
            "✅ 設定を更新しました。",
            ephemeral=True
        )

class ColorModal(discord.ui.Modal):

    def __init__(self, current):
        super().__init__(
            title="パネルカラーを変更"
        )

        self.input = discord.ui.TextInput(
            label="カラーコード",
            placeholder="#58A6FF",
            default=str(current),
            max_length=7,
            required=True
        )

        self.add_item(
            self.input
        )

    async def on_submit(self, interaction):

        value = (
            str(self.input)
            .strip()
            .replace("#", "")
        )

        if not re.fullmatch(
            r"[0-9A-Fa-f]{6}",
            value
        ):
            await interaction.response.send_message(
                "❌ 6桁のカラーコードを入力してください。"
                "例: `#58A6FF`",
                ephemeral=True
            )
            return

        set_guild_value(
            interaction.guild.id,
            "ticket_embed_color",
            value
        )

        await interaction.response.send_message(
            "✅ パネルカラーを変更しました。",
            ephemeral=True
        )

class AutoAnswerAddModal(discord.ui.Modal):

    def __init__(self):
        super().__init__(
            title="自動回答を追加"
        )

        self.keyword = discord.ui.TextInput(
            label="キーワード",
            placeholder="例：IP / 料金 / 参加方法",
            max_length=100,
            required=True
        )

        self.response = discord.ui.TextInput(
            label="自動回答",
            placeholder="このキーワードが含まれたら送信する文章",
            max_length=2000,
            required=True,
            style=discord.TextStyle.paragraph
        )

        self.add_item(
            self.keyword
        )

        self.add_item(
            self.response
        )

    async def on_submit(self, interaction):

        items = auto_answers(
            get_guild_config(
                interaction.guild.id
            )
        )

        keyword = str(
            self.keyword
        ).strip()

        response = str(
            self.response
        ).strip()

        if len(items) >= 25:
            await interaction.response.send_message(
                "❌ 自動回答は最大25個までです。",
                ephemeral=True
            )
            return

        if any(
            item["keyword"].casefold()
            == keyword.casefold()
            for item in items
        ):
            await interaction.response.send_message(
                "❌ そのキーワードはすでに登録されています。",
                ephemeral=True
            )
            return

        items.append({
            "keyword": keyword,
            "response": response
        })

        set_guild_value(
            interaction.guild.id,
            "ticket_auto_answers",
            items
        )

        await interaction.response.send_message(
            f"✅ **{keyword}** の自動回答を追加しました。",
            ephemeral=True
        )


class AutoAnswerEditModal(discord.ui.Modal):

    def __init__(
        self,
        index,
        current_keyword,
        current_response
    ):
        super().__init__(
            title="自動回答を編集"
        )

        self.index = index

        self.keyword = discord.ui.TextInput(
            label="キーワード",
            default=current_keyword[:100],
            max_length=100,
            required=True
        )

        self.response = discord.ui.TextInput(
            label="自動回答",
            default=current_response[:2000],
            max_length=2000,
            required=True,
            style=discord.TextStyle.paragraph
        )

        self.add_item(
            self.keyword
        )

        self.add_item(
            self.response
        )

    async def on_submit(self, interaction):

        items = auto_answers(
            get_guild_config(
                interaction.guild.id
            )
        )

        if (
            self.index < 0
            or self.index >= len(items)
        ):
            await interaction.response.send_message(
                "❌ 自動回答が見つかりません。",
                ephemeral=True
            )
            return

        keyword = str(
            self.keyword
        ).strip()

        response = str(
            self.response
        ).strip()

        if not keyword or not response:
            await interaction.response.send_message(
                "❌ キーワードと回答の両方を入力してください。",
                ephemeral=True
            )
            return

        for index, item in enumerate(items):

            if index == self.index:
                continue

            if (
                item["keyword"].casefold()
                == keyword.casefold()
            ):
                await interaction.response.send_message(
                    "❌ そのキーワードはすでに登録されています。",
                    ephemeral=True
                )
                return

        items[self.index] = {
            "keyword": keyword[:100],
            "response": response[:2000]
        }

        set_guild_value(
            interaction.guild.id,
            "ticket_auto_answers",
            items
        )

        await interaction.response.send_message(
            f"✅ **{keyword}** の自動回答を更新しました。",
            ephemeral=True
        )


class AutoAnswerEditSelect(discord.ui.Select):

    def __init__(self, items):

        options = [
            discord.SelectOption(
                label=item["keyword"][:100],
                value=str(index),
                emoji="✏️",
                description=item["response"][:100]
            )
            for index, item in enumerate(items[:25])
        ]

        super().__init__(
            placeholder="編集する自動回答を選択",
            options=options,
            min_values=1,
            max_values=1
        )

        self.items_snapshot = items

    async def callback(self, interaction):

        try:
            index = int(
                self.values[0]
            )

            item = self.items_snapshot[index]

        except (ValueError, IndexError):
            await interaction.response.send_message(
                "❌ 自動回答の取得に失敗しました。",
                ephemeral=True
            )
            return

        await interaction.response.send_modal(
            AutoAnswerEditModal(
                index,
                item["keyword"],
                item["response"]
            )
        )


class AutoAnswerEditView(discord.ui.View):

    def __init__(self, items):
        super().__init__(timeout=120)
        self.add_item(

class AutoAnswerDeleteSelect(discord.ui.Select):

    def __init__(self, items):

        options = [
            discord.SelectOption(
                label=item["keyword"][:100],
                value=str(index),
                emoji="🗑️"
            )
            for index, item in enumerate(items[:25])
        ]

        super().__init__(
            placeholder="削除する自動回答を選択",
            options=options,
            min_values=1,
            max_values=1
        )

    async def callback(self, interaction):

        items = auto_answers(
            get_guild_config(
                interaction.guild.id
            )
        )

        try:
            removed = items.pop(
                int(self.values[0])
            )

        except (ValueError, IndexError):
            await interaction.response.send_message(
                "❌ 自動回答の取得に失敗しました。",
                ephemeral=True
            )
            return

        set_guild_value(
            interaction.guild.id,
            "ticket_auto_answers",
            items
        )

        await interaction.response.send_message(
            f"🗑️ **{removed['keyword']}** "
            "の自動回答を削除しました。",
            ephemeral=True
        )


class AutoAnswerDeleteView(discord.ui.View):

    def __init__(self, items):
        super().__init__(timeout=120)
        self.add_item(
            AutoAnswerDeleteSelect(items)
        )

class TicketMessageListener(commands.Cog):

    def __init__(self, bot):
        self.bot = bot

    @commands.Cog.listener()
    async def on_message(self, message):

        if (
            message.author.bot
            or not message.guild
            or not isinstance(
                message.channel,
                discord.TextChannel
            )
        ):
            return

        creator, replied = state(
            message.channel.topic
        )

        if not creator:
            return

        config = cfg(
            message.guild.id
        )

        staff_ids = config.get(
            "ticket_staff_role_ids",
            []
        )

        is_staff_message = (
            isinstance(
                message.author,
                discord.Member
            )
            and staff(
                message.author,
                staff_ids
            )
        )

        # -------------------------------------------------
        # 自動回答
        # -------------------------------------------------

        if not is_staff_message:

            content = message.content.strip()

            if content:

                answers = auto_answers(
                    config
                )

                matched = []

                lowered = content.casefold()

                for item in answers:

                    if (
                        item["keyword"].casefold()
                        in lowered
                    ):
                        matched.append(
                            item["response"]
                        )

                if matched:

                    unique = list(
                        dict.fromkeys(
                            matched
                        )
                    )

                    try:
                        await message.channel.send(
                            "\n\n".join(unique)
                        )
                    except discord.HTTPException:
                        pass

        # -------------------------------------------------
        # 最初のメッセージへの自動返信
        # -------------------------------------------------

        if replied:
            return

        if message.author.id != creator:
            return

        if (
            not message.content.strip()
            and not message.attachments
        ):
            return

        if config["ticket_first_reply"].strip():

            try:
                await message.channel.send(
                    replace_placeholders(
                        config["ticket_first_reply"],
                        message.author
                    )
                )

            except discord.HTTPException:
                return

        try:
            await message.channel.edit(
                topic=topic(
                    creator,
                    True
                )
            )
        except discord.HTTPException:
            pass

    await bot.add_cog(

    )

    await bot.add_cog(
